// === 開啟地點彈出視窗 === window.openPlaceModal = function(name, defaultImg, rating, desc) { window.currentModalPlace = name; var finalImg = window.customImages[name] || defaultImg; document.getElementById('modalName').innerText = name; document.getElementById('modalImg').src = finalImg; document.getElementById('modalRating').innerHTML = ' ' + rating; document.getElementById('modalDesc').innerText = desc; // 🌟 真正的 Google Maps 官方路線規劃網址 var mapUrl = 'https://www.google.com/maps/dir/?api=1&destination=' + encodeURIComponent(name); // 針對特定地點，加上更精確的地區關鍵字 if (name === "劉家酸菜白肉鍋") { mapUrl = 'https://www.google.com/maps/dir/?api=1&destination=' + encodeURIComponent("劉家酸菜白肉鍋 中正堂本館"); } else if (name.indexOf("捉鳳凰") !== -1) { mapUrl = 'https://www.google.com/maps/dir/?api=1&destination=' + encodeURIComponent("捉鳳凰百年老宅 台南"); } document.getElementById('modalNavBtn').href = mapUrl; document.getElementById('detailModal').style.display = 'flex'; };
